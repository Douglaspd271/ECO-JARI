rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /denuncias/{id} {
      allow read: if true;
      allow create: if request.resource.size < 200000;
      allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(['done']);
      allow delete: if false;
    }
  }
}
